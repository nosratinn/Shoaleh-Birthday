<script>
const APPS_SCRIPT_URL = "https://script.google.com/macros/s/AKfycbyvAd_y2m5bcGJ_vbC2o0NmTThlJgQ_nB0zJPKwID49JH2mrQ2E1LT1j-oylCCoK7u6Lw/exec";
const PARTY_PIN = "0624";

let mediaRecorder = null;
let recordedChunks = [];
let videoStream = null;
let countdownTimer = null;
let recordingTimer = null;
let blinkTimer = null;
let recordingSeconds = 0;
let isRecording = false;
let currentMimeType = "";

window.addEventListener("load", () => {
if (localStorage.getItem("party_unlocked") === "true") {
startCamera();
showScreen("welcome-screen");
} else {
showScreen("pin-screen");
}
});

function checkPin() {
const input = document.getElementById("pin-input").value;
const errorText = document.getElementById("pin-error");
if (input === PARTY_PIN) {
localStorage.setItem("party_unlocked", "true");
startCamera();
showScreen("welcome-screen");
} else {
errorText.style.opacity = "1";
setTimeout(() => { errorText.style.opacity = "0"; }, 3000);
document.getElementById("pin-input").value = "";
}
}

async function startCamera() {
try {
videoStream = await navigator.mediaDevices.getUserMedia({
video: { facingMode: "user", width: { ideal: 1280 }, height: { ideal: 720 } },
audio: { echoCancellation: true, noiseSuppression: true }
});
document.getElementById("camera-feed").srcObject = videoStream;
} catch (err) {
console.error("Camera error:", err);
showError("Could not access the camera. Please make sure you have allowed camera/microphone access.");
}
}

function showScreen(screenId) {
document.querySelectorAll(".screen").forEach(s => s.classList.remove("active"));
document.getElementById(screenId).classList.add("active");
}

function startCountdown() {
showScreen("countdown-screen");
let count = 10;
document.getElementById("countdown-number").textContent = count;
countdownTimer = setInterval(() => {
count--;
document.getElementById("countdown-number").textContent = count;
if (count <= 0) {
clearInterval(countdownTimer);
setTimeout(() => beginRecording(), 600);
}
}, 1000);
}

function beginRecording() {
showScreen("recording-screen");
recordedChunks = [];
recordingSeconds = 0;
isRecording = true;
if (!videoStream || !videoStream.active) { showError("Camera feed lost. Please refresh the page."); return; }
document.getElementById("warning-box").classList.remove("visible");
document.getElementById("warn-seconds").textContent = "10 seconds remaining";
document.getElementById("rec-timer").textContent = "0:00";
try { mediaRecorder = new MediaRecorder(videoStream, { videoBitsPerSecond: 1500000 }); } catch (e) { mediaRecorder = new MediaRecorder(videoStream); }
currentMimeType = mediaRecorder.mimeType || "video/webm";
mediaRecorder.ondataavailable = (event) => { if (event.data && event.data.size > 0) recordedChunks.push(event.data); };
mediaRecorder.onstop = () => { if (recordedChunks.length > 0) showScreen("review-screen"); else showError("Recording failed: No data captured. Please refresh and try again."); };
mediaRecorder.start(1000);
let dotVisible = true;
blinkTimer = setInterval(() => { dotVisible = !dotVisible; document.getElementById("rec-dot").classList.toggle("blink", !dotVisible); }, 500);
recordingTimer = setInterval(() => {
recordingSeconds++;
const mins = Math.floor(recordingSeconds / 60);
const secs = recordingSeconds % 60;
document.getElementById("rec-timer").textContent = mins + ":" + String(secs).padStart(2, "0");
if (recordingSeconds === 50) document.getElementById("warning-box").classList.add("visible");
if (recordingSeconds >= 50) document.getElementById("warn-seconds").textContent = (60 - recordingSeconds) + " seconds remaining";
if (recordingSeconds >= 60) finishRecording();
}, 1000);
}

function finishRecording() {
if (!isRecording) return;
isRecording = false;
clearInterval(recordingTimer);
clearInterval(blinkTimer);
recordingTimer = null;
blinkTimer = null;
if (mediaRecorder && mediaRecorder.state !== "inactive") { mediaRecorder.stop(); } else { showScreen("review-screen"); }
}

function saveVideo() {
showScreen("uploading-screen");
document.getElementById("upload-status").textContent = "Saving your message...";
uploadToGoogleDrive();
}

function recordAgain() { recordedChunks = []; resetToWelcome(); }

async function uploadToGoogleDrive() {
try {
const finalType = currentMimeType;
const cleanMimeType = finalType.split(';') || "video/webm";
const blob = new Blob(recordedChunks, { type: finalType });
if (blob.size === 0) throw new Error("Video has no data");
const base64 = await blobToBase64(blob);
const base64Data = base64.split(",");

const CHUNK_SIZE = 3 * 1024 * 1024;
const totalChunks = Math.ceil(base64Data.length / CHUNK_SIZE);
const sessionId = Date.now() + "_" + Math.random().toString(36).substr(2, 9);
const statusEl = document.getElementById("upload-status");
for (let i = 0; i < totalChunks; i++) {
statusEl.textContent = "Saving... (" + (i + 1) + "/" + totalChunks + ")";
const start = i * CHUNK_SIZE;
const end = Math.min(start + CHUNK_SIZE, base64Data.length);
const chunk = base64Data.substring(start, end);
const controller = new AbortController();
const timeout = setTimeout(() => controller.abort(), 120000);
await fetch(APPS_SCRIPT_URL, {
method: "POST",
mode: "no-cors",
headers: { "Content-Type": "text/plain" },
signal: controller.signal,
body: JSON.stringify({ sessionId: sessionId, chunkIndex: i, totalChunks: totalChunks, mimeType: cleanMimeType, data: chunk })
});
clearTimeout(timeout);
}
statusEl.textContent = "Saving your message...";
recordedChunks = [];
showThankYou();
} catch (err) {
document.getElementById("upload-status").textContent = "Saving your message...";
showError("There was a problem saving your video. Please try again.");
}
}

function blobToBase64(blob) {
return new Promise((resolve, reject) => {
const reader = new FileReader();
reader.onloadend = () => resolve(reader.result);
reader.onerror = reject;
reader.readAsDataURL(blob);
});
}

function showThankYou() { showScreen("thankyou-screen"); setTimeout(() => resetToWelcome(), 5000); }
function showError(message) { document.getElementById("error-message").textContent = message; showScreen("error-screen"); }

function resetToWelcome() {
clearInterval(countdownTimer); clearInterval(recordingTimer); clearInterval(blinkTimer);
countdownTimer = null; recordingTimer = null; blinkTimer = null; isRecording = false; recordingSeconds = 0; recordedChunks = [];
if (mediaRecorder && mediaRecorder.state !== "inactive") mediaRecorder.stop();
mediaRecorder = null;
document.getElementById("rec-dot").classList.remove("blink");
showScreen("welcome-screen");
}
